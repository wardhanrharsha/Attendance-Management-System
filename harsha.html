import java.util.Scanner; public class Fibonacci { static int fabRec(int n) { if (n <= 1) return n; else return fabRec(n - 1) + fabRec(n - 2); } static void fabNrec(int n) { int first = 0, second = 1; for (int i = 0; i < n; i++) { System.out.print(first + " "); int next = first + second; first = second; second = next; } } public static void main(String[] args) { Scanner sc = new Scanner(System.in); System.out.print("Enter number of terms: "); int n = sc.nextInt(); System.out.println("Fibonacci using Recursion:"); for (int i = 0; i < n; i++) { System.out.println(fabRec(i)); } System.out.println("\nFibonacci without Recursion:"); fabNrec(n); sc.close(); } }
